# Øvelse del 5: Pseudokode
```vbnet
BEGIN TilmeldElevTilFag(studentId, subjectId)
    
    # Trin 1: Hent data
    SET elev TO Database.GetStudent(studentId)
    SET fag TO Database.GetSubject(subjectId)

    # Trin 2: Validering af forretningsregler
    IF elev.RegisteredSubjectsCount >= 5 THEN
        RETURN "Fejl: Eleven har allerede max antal fag (5)"
    END IF

    IF fag.RegisteredStudentsCount >= 30 THEN
        RETURN "Fejl: Faget er fuldt (Max 30 elever)"
    END IF

    # Trin 3: Gem hvis alt er OK
    Database.SaveRegistration(studentId, subjectId)
    
    RETURN "Succes: Elev er tilmeldt faget"

END
```
